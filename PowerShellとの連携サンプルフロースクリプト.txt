System.RunApplication.RunApplication ApplicationPath: $'''powershell''' CommandLineArguments: $'''-ExecutionPolicy Bypass -Command \"D:\\PAD\\LogWrite\\Log_Write.ps1\"''' WindowStyle: System.ProcessWindowStyle.Normal
LOOP LoopIndex FROM 0 TO 10 STEP 1
    /# SendMessageに任意の値を代入してスクリプトに渡す
このサンプルではループ回数の二乗を代入#/
    SET SendMessage TO $'''%LoopIndex% * %LoopIndex%'''
    Scripting.RunPowershellScript.RunScript Script: $'''$outputMessage = %SendMessage%
# exit を送ることでログ表示窓を閉じることができる


# 以下は送信処理
$pipeName = \"MyPowerShellPipe\"

# パイプクライアントの作成（送信専用）
$pipeClient = New-Object System.IO.Pipes.NamedPipeClientStream(\".\", $pipeName, [System.IO.Pipes.PipeDirection]::Out)

# サーバーに接続（タイムアウトを3000ミリ秒に設定）
$pipeClient.Connect(3000)

# メッセージを送信
$writer = New-Object System.IO.StreamWriter($pipeClient)
$writer.AutoFlush = $true
$writer.WriteLine($outputMessage)

# 後片付け
$writer.Dispose()
$pipeClient.Dispose()
Write-Host \"送信完了\"'''
